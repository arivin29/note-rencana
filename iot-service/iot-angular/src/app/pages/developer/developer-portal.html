Code Examples
cURL
curl -H "X-API-Key: YOUR_KEY" \
  {{ baseUrl }}/external-api/v1/nodes?status=active&limit=10
Python
import requests

API_KEY = "your_key"
BASE = "{{ baseUrl }}/external-api/v1"
headers = {{ '{' }}"X-API-Key": API_KEY{{ '}' }}

nodes = requests.get(f"{{ '{' }}BASE{{ '}' }}/nodes", headers=headers).json()
print(f"Found {{ '{' }}nodes['total']{{ '}' }} nodes")
JavaScript (fetch)
const API_KEY = 'your_key';
const BASE = '{{ baseUrl }}/external-api/v1';
const headers = {{ '{' }} 'X-API-Key': API_KEY {{ '}' }};

const res = await fetch(`${{ '{' }}BASE{{ '}' }}/sensor-data/latest?sensorIds=uuid1`, {{ '{' }} headers {{ '}' }});
const data = await res.json();
console.log(data);
